cmake_minimum_required(VERSION 3.16)
project(lidar_workbench)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/src/coin_d4_node_handler.cpp")
  message(FATAL_ERROR "Build through tools/lidar_workbench/build.sh to stage the verified driver")
endif()
add_library(workbench_sensor_sdk SHARED
  vendor/src/lidar_sdk/lidar_data_processor.cpp vendor/src/lidar_sdk/mtime.cpp
  vendor/src/lidar_sdk/serial_port.cpp vendor/src/lidar_sdk/timer.cpp
  vendor/src/coin_d4_base_handler.cpp vendor/src/coin_d4_node_handler.cpp)
target_include_directories(workbench_sensor_sdk PUBLIC vendor/include)
target_link_libraries(workbench_sensor_sdk rclcpp::rclcpp ${sensor_msgs_TARGETS} ${std_msgs_TARGETS})
function(workbench_target name source)
  add_executable(${name} ${source})
  target_include_directories(${name} PRIVATE vendor/include)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(${name} rclcpp::rclcpp ${sensor_msgs_TARGETS} ${nav_msgs_TARGETS} ${std_msgs_TARGETS})
endfunction()
workbench_target(workbench_sensor sensor_node.cpp)
target_link_libraries(workbench_sensor workbench_sensor_sdk)
workbench_target(workbench_fusion fusion_node.cpp)
target_link_libraries(workbench_fusion tf2_ros::tf2_ros)
workbench_target(lidar_workbench workbench.cpp)
target_link_libraries(lidar_workbench ament_index_cpp::ament_index_cpp yaml-cpp
  ${geometry_msgs_TARGETS} ${tf2_msgs_TARGETS} ${slam_toolbox_TARGETS})
workbench_target(verify_mock_recording verify_mock_recording.cpp)
target_link_libraries(verify_mock_recording rosbag2_cpp::rosbag2_cpp yaml-cpp ${geometry_msgs_TARGETS})
install(TARGETS workbench_sensor_sdk LIBRARY DESTINATION lib)
install(TARGETS workbench_sensor workbench_fusion lidar_workbench DESTINATION lib/${PROJECT_NAME})
install(TARGETS verify_mock_recording DESTINATION lib/${PROJECT_NAME})
install(FILES record_qos.yaml workbench.rviz DESTINATION share/${PROJECT_NAME})
install(FILES vendor/LICENSE DESTINATION share/${PROJECT_NAME} RENAME DRIVER_LICENSE)
if(BUILD_TESTING)
  enable_testing()
  workbench_target(test_workbench test_workbench.cpp)
  target_compile_options(test_workbench PRIVATE -UNDEBUG)
  add_test(NAME workbench_core COMMAND test_workbench)
endif()
ament_package()
