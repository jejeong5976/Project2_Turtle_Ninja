cmake_minimum_required(VERSION 3.16)
project(nt_timing_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_executable(inspect_trace inspect_trace.cpp)
target_link_libraries(inspect_trace Threads::Threads)
add_executable(test_trace test_trace.cpp)
target_link_libraries(test_trace Threads::Threads)
enable_testing()
add_test(NAME trace COMMAND test_trace)
option(WITH_ROS "Build passive observer" ON)
if(WITH_ROS)
 find_package(rclcpp REQUIRED)
 foreach(dep nav_msgs sensor_msgs tf2_msgs)
  find_package(${dep} REQUIRED)
 endforeach()
 add_executable(observe_timing observe_timing.cpp)
 target_link_libraries(observe_timing rclcpp::rclcpp ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS} ${tf2_msgs_TARGETS} Threads::Threads)
endif()
