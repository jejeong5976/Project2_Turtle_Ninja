cmake_minimum_required(VERSION 3.16)
project(scanned_map_controller_sim LANGUAGES CXX)
find_package(rclcpp REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_map_server REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(yaml-cpp REQUIRED)
if(NOT BODY_ROUTE_PREFIX)
  message(FATAL_ERROR "Supply BODY_ROUTE_PREFIX from the main-based build")
endif()
foreach(target prepare prepare_unknown import_scenario unknown_suite feed evaluate startup)
  add_executable(${target} ${target}.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_include_directories(${target} SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS} ${nav2_costmap_2d_INCLUDE_DIRS} ${nav2_map_server_INCLUDE_DIRS})
  target_include_directories(${target} PRIVATE ${BODY_ROUTE_PREFIX}/include)
  target_link_libraries(${target} ${BODY_ROUTE_PREFIX}/lib/libbody_route_plugin.so ${OpenCV_LIBS} yaml-cpp
    ${nav2_costmap_2d_LIBRARIES} ${geometry_msgs_TARGETS} ${nav_msgs_TARGETS} ${std_msgs_TARGETS})
endforeach()
target_link_libraries(prepare ${nav2_map_server_LIBRARIES})
target_link_libraries(feed rclcpp::rclcpp tf2_ros::tf2_ros ${sensor_msgs_TARGETS} ${nav2_msgs_TARGETS})

target_link_libraries(startup rclcpp::rclcpp tf2_ros::tf2_ros ${nav2_msgs_TARGETS} ${lifecycle_msgs_TARGETS})

target_include_directories(unknown_suite PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../ros2_ws/src/body_route_controller/test)
