cmake_minimum_required(VERSION 3.16)
project(mapping_resolution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
foreach(dependency rclcpp rclcpp_lifecycle rosbag2_cpp nav_msgs sensor_msgs tf2_msgs)
  find_package(${dependency} REQUIRED)
endforeach()
find_package(Boost REQUIRED COMPONENTS serialization)
find_package(Eigen3 REQUIRED)
find_package(TBB REQUIRED)
find_path(KARTO_INCLUDE karto_sdk/Karto.h HINTS "${rclcpp_DIR}/../../../include" REQUIRED)
find_library(KARTO_LIBRARY kartoSlamToolbox HINTS "${rclcpp_DIR}/../../../lib" REQUIRED)
add_executable(compare_mapping_resolution ../compare_mapping_resolution.cpp)
target_include_directories(compare_mapping_resolution SYSTEM PRIVATE ${KARTO_INCLUDE})
target_compile_options(compare_mapping_resolution PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(compare_mapping_resolution rclcpp::rclcpp rosbag2_cpp::rosbag2_cpp ${KARTO_LIBRARY} Boost::serialization
  ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS} ${tf2_msgs_TARGETS})
add_executable(render_mapping_graph ../render_mapping_graph.cpp)
target_include_directories(render_mapping_graph SYSTEM PRIVATE ${KARTO_INCLUDE})
target_compile_options(render_mapping_graph PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(render_mapping_graph ${KARTO_LIBRARY} Boost::serialization Eigen3::Eigen TBB::tbb rclcpp_lifecycle::rclcpp_lifecycle)
