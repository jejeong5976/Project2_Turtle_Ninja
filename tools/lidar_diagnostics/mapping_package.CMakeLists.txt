cmake_minimum_required(VERSION 3.16)
project(coin_d4_mapping_driver)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(mapping_sensor SHARED src/lidar_sdk/lidar_data_processor.cpp
  src/lidar_sdk/mtime.cpp src/lidar_sdk/serial_port.cpp src/lidar_sdk/timer.cpp
  src/coin_d4_base_handler.cpp src/coin_d4_node_handler.cpp)
target_include_directories(mapping_sensor PUBLIC include)
target_link_libraries(mapping_sensor rclcpp::rclcpp ${sensor_msgs_TARGETS} ${std_msgs_TARGETS})
add_executable(front_mapping_input src/mapping_node.cpp)
target_compile_options(front_mapping_input PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(front_mapping_input mapping_sensor ${nav_msgs_TARGETS})
install(TARGETS mapping_sensor LIBRARY DESTINATION lib)
install(TARGETS front_mapping_input DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_mapping_node src/test_mapping_node.cpp)
  target_compile_options(test_mapping_node PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(test_mapping_node mapping_sensor ${nav_msgs_TARGETS})
  add_executable(test_mapping_parser src/test_timing_parser.cpp)
  target_compile_options(test_mapping_parser PRIVATE -UNDEBUG)
  target_link_libraries(test_mapping_parser mapping_sensor util)
endif()
ament_package()
