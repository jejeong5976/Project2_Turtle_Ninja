add_library(body_route_core STATIC src/core.cpp src/supervision.cpp)
set_target_properties(body_route_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(body_route_core PUBLIC cxx_std_17)
target_include_directories(body_route_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(body_route_core PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_library(body_route_plugin SHARED src/controller.cpp src/collision.cpp src/maze_planner.cpp src/observation.cpp src/observation_layer.cpp src/observation_controller.cpp)
target_compile_features(body_route_plugin PUBLIC cxx_std_17)
target_compile_options(body_route_plugin PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(body_route_plugin PUBLIC body_route_core
  ${nav2_controller_LIBRARIES} ${nav2_core_LIBRARIES} ${nav2_costmap_2d_LIBRARIES} ${nav2_ros_common_LIBRARIES}
  pluginlib::pluginlib rclcpp::rclcpp tf2::tf2 tf2_geometry_msgs::tf2_geometry_msgs
  ${geometry_msgs_TARGETS} ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS} ${std_msgs_TARGETS})
target_include_directories(body_route_plugin SYSTEM PUBLIC
  ${nav2_controller_INCLUDE_DIRS} ${nav2_core_INCLUDE_DIRS} ${nav2_costmap_2d_INCLUDE_DIRS} ${nav2_ros_common_INCLUDE_DIRS})
add_library(trial_profile STATIC src/profile.cpp)

target_include_directories(trial_profile PUBLIC include)

target_compile_options(configure_trial PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(route_trial src/route_trial.cpp)
target_compile_features(route_trial PRIVATE cxx_std_17)
target_compile_options(route_trial PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(route_trial trial_profile body_route_core rclcpp::rclcpp rclcpp_action::rclcpp_action
  tf2_ros::tf2_ros ${nav2_msgs_TARGETS} ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS}
  ${geometry_msgs_TARGETS} ${std_msgs_TARGETS})
install(TARGETS route_trial configure_trial DESTINATION lib/${PROJECT_NAME})
add_library(wall_layout STATIC src/layout.cpp src/layout_navigation.cpp)
target_compile_features(wall_layout PUBLIC cxx_std_17)
target_include_directories(wall_layout PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)

target_link_libraries(capture_layout trial_profile body_route_core rclcpp::rclcpp tf2_ros::tf2_ros
  ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS})
