cmake_minimum_required(VERSION 3.8)
project(robot_cli)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(action_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(interfaces REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(urdf REQUIRED)
find_package(yaml-cpp REQUIRED)
set(CURSES_NEED_WIDE TRUE)
find_package(Curses REQUIRED)

# dice_map.yaml 은 설치 안 됨 → mission 과 같이 레포 maps/ 를 컴파일 때 박음. 다른 곳이면 -DROBOT_CLI_MAPS_DIR=<경로>
set(ROBOT_CLI_MAPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../maps" CACHE PATH "dice_map.yaml 이 있는 폴더")
get_filename_component(ROBOT_CLI_MAPS_DIR "${ROBOT_CLI_MAPS_DIR}" ABSOLUTE)

# ROS 없는 부분 (설정 · 명령 문법 · 상태 글 · 실험 기록) — 시험이 이것만 씀
add_library(robot_cli_core STATIC src/model.cpp src/command.cpp src/status_view.cpp src/pages.cpp src/teach.cpp src/bag.cpp)
target_include_directories(robot_cli_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(robot_cli_core PUBLIC cxx_std_17)
target_link_libraries(robot_cli_core yaml-cpp::yaml-cpp ${interfaces_TARGETS})

add_executable(robot_cli src/main.cpp src/robot_link.cpp src/tui.cpp)
target_include_directories(robot_cli PRIVATE ${CURSES_INCLUDE_DIRS})
target_compile_definitions(robot_cli PRIVATE ROBOT_CLI_MAPS_DIR="${ROBOT_CLI_MAPS_DIR}")
target_link_libraries(robot_cli robot_cli_core rclcpp::rclcpp rclcpp_action::rclcpp_action
  ament_index_cpp::ament_index_cpp tf2::tf2 tf2_ros::tf2_ros ${CURSES_LIBRARIES}
  ${action_msgs_TARGETS} ${controller_manager_msgs_TARGETS} ${diagnostic_msgs_TARGETS} ${geometry_msgs_TARGETS}
  ${nav_msgs_TARGETS} ${nav2_msgs_TARGETS} ${sensor_msgs_TARGETS} ${std_msgs_TARGETS} ${std_srvs_TARGETS}
  ${interfaces_TARGETS} ${rcl_interfaces_TARGETS} urdf::urdf)
install(TARGETS robot_cli RUNTIME DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  # 레포 설정 파일을 그대로 읽어 봄 (bringup 이 설치하는 config/ 원본)
  set(ROBOT_CLI_CONFIG_DIR "${ROBOT_CLI_MAPS_DIR}/../config" CACHE PATH "hardware_map.yaml · rack_cells.yaml 폴더")
  ament_add_gtest(test_model test/test_model.cpp)
  target_link_libraries(test_model robot_cli_core)
  target_compile_definitions(test_model PRIVATE TEST_MAPS_DIR="${ROBOT_CLI_MAPS_DIR}" TEST_CONFIG_DIR="${ROBOT_CLI_CONFIG_DIR}")
  ament_add_gtest(test_command test/test_command.cpp)
  target_link_libraries(test_command robot_cli_core)
  ament_add_gtest(test_status_view test/test_status_view.cpp)
  target_link_libraries(test_status_view robot_cli_core)
  ament_add_gtest(test_teach test/test_teach.cpp)
  target_link_libraries(test_teach robot_cli_core)
endif()
ament_package()
