cmake_minimum_required(VERSION 3.10)
project(manipulation)
# 빌드 종류를 안 주면 Release: 최적화 없이 빌드하면 PICK · PLACE 의 IK 가 수 초 걸려 그동안 감시 · 심장 박동이 멈춤 (O2 는 0.1 s 안)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
# aarch64 GCC: std::pair 인자 ABI 가 GCC 10.1 에서 바뀌었다는 안내 (-Wpsabi). 모두 같은 GCC 로 빌드하므로 끔
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  add_compile_options(-Wno-psabi)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)   # 쉬는 팔 요청 · 팔 보드 토크 상태 (std_msgs/Bool)
find_package(geometry_msgs REQUIRED)
find_package(OpenCV REQUIRED)
add_library(manipulation_core STATIC src/arm.cpp src/arm_path.cpp src/arm_kinematics.cpp src/arm_state.cpp src/grasp_select.cpp src/rack_tags.cpp src/standup_table.cpp src/park_stand.cpp
  src/cell_shift.cpp src/camera_drift.cpp src/held_correction.cpp src/look_variant.cpp src/visual_regrasp.cpp src/grasp_guard.cpp
  src/held_image.cpp src/arm_rest.cpp src/placed_memory.cpp src/npz_policy.cpp src/nudge.cpp
  src/look_at.cpp)
set_target_properties(manipulation_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(manipulation_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(manipulation_core PUBLIC cxx_std_17)
target_link_libraries(manipulation_core Eigen3::Eigen)
# npz 정책 (nudge): np.savez (stored) 는 zlib 없이 읽음. zlib 이 있으면 np.savez_compressed (deflate) 도 — 없어도 빌드는 됨
find_package(ZLIB QUIET)
if(ZLIB_FOUND)
  target_compile_definitions(manipulation_core PUBLIC MANIP_HAVE_ZLIB)
  target_link_libraries(manipulation_core ZLIB::ZLIB)
endif()
# 팔 작업 · 차례 · 적재함 칸 · 쥔 퍽 태그 (rack_tag_stage: perception /perception/wrist/tags → 칸 바닥 평면 · PnP)
add_executable(control_master src/control_master.cpp src/rack_tag_stage.cpp)
target_include_directories(control_master PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(control_master manipulation_core rclcpp::rclcpp rclcpp_action::rclcpp_action
  ${control_msgs_TARGETS} ${trajectory_msgs_TARGETS} ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS}
  ${controller_manager_msgs_TARGETS} ${lifecycle_msgs_TARGETS} ${interfaces_TARGETS} ${std_srvs_TARGETS} ${std_msgs_TARGETS}
  tf2::tf2 tf2_ros::tf2_ros tf2_geometry_msgs::tf2_geometry_msgs ${OpenCV_LIBS})
add_executable(arm_state_publisher src/arm_state_publisher.cpp)
target_link_libraries(arm_state_publisher manipulation_core rclcpp::rclcpp ${sensor_msgs_TARGETS})
# 홈 자세 손목 사진으로 카메라가 돈 만큼 → /wrist_cam/drift (control_master 가 손목 검출을 되돌림)
add_executable(camera_drift_node src/camera_drift_node.cpp)
target_include_directories(camera_drift_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(camera_drift_node manipulation_core rclcpp::rclcpp ${sensor_msgs_TARGETS} ${std_srvs_TARGETS} ${geometry_msgs_TARGETS}
  ${OpenCV_LIBS})
install(TARGETS manipulation_core control_master arm_state_publisher camera_drift_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
