cmake_minimum_required(VERSION 3.10)
project(manipulation)
# 빌드 종류를 안 주면 Release: 최적화 없이 빌드하면 PICK · PLACE 의 IK 가 수 초 걸려 그동안 감시 · 심장 박동이 멈춤 (O2 는 0.1 s 안)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
# aarch64 GCC: std::pair 인자 ABI 가 GCC 10.1 에서 바뀌었다는 안내 (-Wpsabi). 모두 같은 GCC 로 빌드하므로 끔
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  add_compile_options(-Wno-psabi)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(OpenCV REQUIRED)
add_library(manipulation_core STATIC src/arm.cpp src/arm_path.cpp src/arm_kinematics.cpp src/arm_state.cpp src/grasp_select.cpp src/rack_tags.cpp src/standup_table.cpp src/park_stand.cpp
  src/cell_shift.cpp src/camera_drift.cpp)
set_target_properties(manipulation_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(manipulation_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(manipulation_core PUBLIC cxx_std_17)
target_link_libraries(manipulation_core Eigen3::Eigen)
add_executable(control_master src/control_master.cpp)
target_link_libraries(control_master manipulation_core rclcpp::rclcpp rclcpp_action::rclcpp_action
  ${control_msgs_TARGETS} ${trajectory_msgs_TARGETS} ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS}
  ${controller_manager_msgs_TARGETS} ${lifecycle_msgs_TARGETS} ${interfaces_TARGETS} ${std_srvs_TARGETS}
  tf2::tf2 tf2_ros::tf2_ros tf2_geometry_msgs::tf2_geometry_msgs)
add_executable(arm_state_publisher src/arm_state_publisher.cpp)
target_link_libraries(arm_state_publisher manipulation_core rclcpp::rclcpp ${sensor_msgs_TARGETS})
# 칸 바닥 AprilTag 로 칸 위치 재기 (팔은 안 움직임). apriltag 없을 때 OpenCV aruco: 4.6 은 contrib, 4.7+ 는 objdetect
add_executable(rack_tag_node src/rack_tag_node.cpp)
target_include_directories(rack_tag_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(rack_tag_node manipulation_core rclcpp::rclcpp ${sensor_msgs_TARGETS} ${std_srvs_TARGETS}
  ${interfaces_TARGETS} ${OpenCV_LIBS} tf2::tf2 tf2_ros::tf2_ros)
# AprilTag 3 (ros-lyrical-apriltag) 있으면 씀: 그림자 진 컵 바닥 태그를 OpenCV aruco 보다 잘 읽음 (실물 스캔 12 칸 ↔ 2 칸)
find_package(apriltag QUIET)
if(apriltag_FOUND)
  target_link_libraries(rack_tag_node apriltag::apriltag)
  target_compile_definitions(rack_tag_node PRIVATE RACK_TAG_HAVE_APRILTAG)
else()
  message(WARNING "apriltag 없음 → 칸 태그는 OpenCV aruco (그림자 진 칸을 덜 읽음). sudo apt install ros-lyrical-apriltag")
endif()
# 홈 자세 손목 사진으로 카메라가 돈 만큼 → /wrist_cam/drift (control_master 가 손목 검출을 되돌림)
add_executable(camera_drift_node src/camera_drift_node.cpp)
target_include_directories(camera_drift_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(camera_drift_node manipulation_core rclcpp::rclcpp ${sensor_msgs_TARGETS} ${std_srvs_TARGETS} ${geometry_msgs_TARGETS}
  ${OpenCV_LIBS})
install(TARGETS manipulation_core control_master arm_state_publisher rack_tag_node camera_drift_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_arm test/test_arm.cpp)
  target_link_libraries(test_arm manipulation_core)
  ament_add_gtest(test_arm_path test/test_arm_path.cpp)
  target_link_libraries(test_arm_path manipulation_core)
  target_compile_definitions(test_arm_path PRIVATE ARM_PATHS_YAML="${CMAKE_CURRENT_SOURCE_DIR}/../../../config/arm_paths.yaml")
  ament_add_gtest(test_camera_drift test/test_camera_drift.cpp)
  target_link_libraries(test_camera_drift manipulation_core)
  ament_add_gtest(test_standup_table test/test_standup_table.cpp)
  target_link_libraries(test_standup_table manipulation_core)
  ament_add_gtest(test_arm_kinematics test/test_arm_kinematics.cpp)
  target_link_libraries(test_arm_kinematics manipulation_core)
  ament_add_gtest(test_arm_state test/test_arm_state.cpp)
  target_link_libraries(test_arm_state manipulation_core)
  ament_add_gtest(test_grasp_select test/test_grasp_select.cpp)
  target_link_libraries(test_grasp_select manipulation_core)
  ament_add_gtest(test_rack_tags test/test_rack_tags.cpp)
  target_link_libraries(test_rack_tags manipulation_core)
  ament_add_gtest(test_cell_shift test/test_cell_shift.cpp)
  target_link_libraries(test_cell_shift manipulation_core)
  ament_add_gtest(test_park_stand test/test_park_stand.cpp TIMEOUT 300)
  target_link_libraries(test_park_stand manipulation_core)
endif()
ament_package()
