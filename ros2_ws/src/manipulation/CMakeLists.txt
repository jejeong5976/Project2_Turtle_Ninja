cmake_minimum_required(VERSION 3.8)
project(manipulation)
# 빌드 종류를 안 주면 Release: 최적화 없이 빌드하면 PICK · PLACE 의 IK 가 수 초 걸려 그동안 감시 · 심장 박동이 멈춤 (O2 는 0.1 s 안)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(manipulation_core STATIC src/arm.cpp src/arm_path.cpp src/arm_kinematics.cpp src/arm_state.cpp src/grasp_select.cpp)
set_target_properties(manipulation_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(manipulation_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(manipulation_core PUBLIC cxx_std_17)
target_link_libraries(manipulation_core Eigen3::Eigen)
add_executable(control_master src/control_master.cpp)
target_link_libraries(control_master manipulation_core rclcpp::rclcpp rclcpp_action::rclcpp_action
  ${control_msgs_TARGETS} ${trajectory_msgs_TARGETS} ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS}
  ${controller_manager_msgs_TARGETS} ${lifecycle_msgs_TARGETS} ${interfaces_TARGETS}
  tf2::tf2 tf2_ros::tf2_ros tf2_geometry_msgs::tf2_geometry_msgs)
add_executable(arm_state_publisher src/arm_state_publisher.cpp)
target_link_libraries(arm_state_publisher manipulation_core rclcpp::rclcpp ${sensor_msgs_TARGETS})
install(TARGETS manipulation_core control_master arm_state_publisher ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_arm test/test_arm.cpp)
  target_link_libraries(test_arm manipulation_core)
  ament_add_gtest(test_arm_path test/test_arm_path.cpp)
  target_link_libraries(test_arm_path manipulation_core)
  ament_add_gtest(test_arm_kinematics test/test_arm_kinematics.cpp)
  target_link_libraries(test_arm_kinematics manipulation_core)
  ament_add_gtest(test_arm_state test/test_arm_state.cpp)
  target_link_libraries(test_arm_state manipulation_core)
  ament_add_gtest(test_grasp_select test/test_grasp_select.cpp)
  target_link_libraries(test_grasp_select manipulation_core)
endif()
ament_package()
