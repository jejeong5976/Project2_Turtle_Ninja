cmake_minimum_required(VERSION 3.16)
project(lidar_cloud_merge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(cloud_core INTERFACE)
target_compile_features(cloud_core INTERFACE cxx_std_17)
target_include_directories(cloud_core INTERFACE include)
target_link_libraries(cloud_core INTERFACE ${sensor_msgs_TARGETS} ${geometry_msgs_TARGETS} tf2::tf2)
add_executable(stationary_cloud_merge src/stationary_cloud_merge.cpp)
target_link_libraries(stationary_cloud_merge cloud_core rclcpp::rclcpp tf2_ros::tf2_ros)
install(TARGETS stationary_cloud_merge DESTINATION lib/${PROJECT_NAME})
add_executable(stationary_front_scan_filter src/stationary_front_scan_filter.cpp)
target_link_libraries(stationary_front_scan_filter cloud_core rclcpp::rclcpp)
install(TARGETS stationary_front_scan_filter DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_cloud_core test/test_cloud_core.cpp)
  target_link_libraries(test_cloud_core cloud_core)
  add_test(NAME cloud_core COMMAND test_cloud_core)
  add_executable(test_scan_filter test/test_scan_filter.cpp)
  target_link_libraries(test_scan_filter cloud_core)
  add_test(NAME scan_filter COMMAND test_scan_filter)
endif()
ament_package()
