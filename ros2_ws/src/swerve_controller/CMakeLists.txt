cmake_minimum_required(VERSION 3.10)
project(swerve_controller)
# realtime_tools 가 부르는 find_package(Boost): CMake 3.30+ 의 CMP0167 경고 → 지금 동작 (FindBoost) 그대로
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()

# 빌드 유형을 안 주면 최적화(Release) — 측정: 100 Hz 주기 계산이 -O0 대비 3~4 배 빠름 (최적화 없음이 기본이었음)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(interfaces REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
add_library(swerve_core STATIC src/swerve.cpp)
set_target_properties(swerve_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(swerve_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(swerve_core PUBLIC cxx_std_17)
add_library(swerve_controller_plugin SHARED src/swerve_controller.cpp)
target_link_libraries(swerve_controller_plugin PUBLIC swerve_core
  controller_interface::controller_interface pluginlib::pluginlib
  rclcpp::rclcpp rclcpp_lifecycle::rclcpp_lifecycle tf2_ros::tf2_ros
  ${geometry_msgs_TARGETS} ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS} ${interfaces_TARGETS})
pluginlib_export_plugin_description_file(controller_interface swerve_controller_plugins.xml)
install(TARGETS swerve_core swerve_controller_plugin ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_swerve test/test_swerve.cpp test/test_blend.cpp)
  target_link_libraries(test_swerve swerve_core)
  ament_add_gtest(test_swerve_controller test/test_swerve_controller.cpp)
  target_link_libraries(test_swerve_controller controller_interface::controller_interface
    pluginlib::pluginlib rclcpp::rclcpp rclcpp_lifecycle::rclcpp_lifecycle ${geometry_msgs_TARGETS})
endif()
ament_package()
